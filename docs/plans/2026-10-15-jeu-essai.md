# Jeu d'essai du pipeline : plan d'implémentation

> **Pour les agents :** sous-compétence requise : superpowers:subagent-driven-development (recommandé) ou superpowers:executing-plans, tâche par tâche. Les étapes se cochent (`- [ ]`).

**But :** faire passer toujours la même entreprise fictive dans le vrai Leon, noter le résultat et garder l'historique du coût et des notes de chaque passage.

**Architecture :** une table `llm_usage` enregistre le coût de chaque appel par organisation et par run, via une enveloppe du client LLM posée par le worker. Un script `pnpm essai` (`src/essai/`) vérifie le jeu, remet à zéro l'organisation d'essai avec le rôle propriétaire, envoie les conversations par la vraie API `ingest`, puis note la mémoire produite avec un juge Haiku et des calculs purs. Le jeu de données vit dans `essai/`, l'historique dans `docs/etudes/essais-pipeline/`.

**Pile :** TypeScript, Hono, Drizzle, Postgres (RLS), pg-boss, SDK Anthropic, zod, Vitest, tsx.

Spec : [`docs/specs/2026-10-15-jeu-essai.md`](../specs/2026-10-15-jeu-essai.md).

```sql
-- 0007-llm-usage.sql : coût de chaque appel au LLM, par organisation et par run.
-- Sert au jeu d'essai (coût mesuré d'un passage) et à savoir ce que coûte chaque client.
-- Le rôle de l'app ne fait qu'insérer ; la lecture passe par le rôle propriétaire.
-- ingest_run_id n'a pas de clé étrangère : la ligne garde le coût d'un run même après sa suppression.

create table llm_usage (
  id uuid primary key default gen_random_uuid(),
  tenant_id uuid not null references organization(id) on delete cascade,
  ingest_run_id uuid,
  step text not null check (step in ('triage', 'extraction', 'placement')),
  model text not null,
  input_tokens integer not null,
  output_tokens integer not null,
  cache_read_tokens integer not null,
  cache_write_tokens integer not null,
  cost_usd numeric(12, 8),
  created_at timestamptz not null default now()
);
create index llm_usage_tenant_idx on llm_usage (tenant_id, created_at desc);

alter table llm_usage enable row level security;
create policy tenant_insert on llm_usage for insert to leon_app
  with check (tenant_id = app_tenant());

grant insert on llm_usage to leon_app;
```

```ts
// Compteurs et coût estimé d'un appel, sans arrondi : la somme de centaines d'appels doit rester juste.
export function usageOf(step: LlmStep, model: string, usage: Usage): LlmUsage {
  const cacheReadTokens = usage.cache_read_input_tokens ?? 0;
  const cacheWriteTokens = usage.cache_creation_input_tokens ?? 0;
  const price = PRICE_PER_MTOK[model];
  const costUsd = price
    ? (usage.input_tokens * price.input +
        usage.output_tokens * price.output +
        cacheReadTokens * price.cacheRead +
        cacheWriteTokens * price.cacheWrite) /
      1_000_000
    : null;
  return {
    step,
    model,
    inputTokens: usage.input_tokens,
    outputTokens: usage.output_tokens,
    cacheReadTokens,
    cacheWriteTokens,
    costUsd,
  };
}

// Journalise les compteurs de tokens et le coût estimé d'un appel, jamais son contenu.
export function logUsage(step: LlmStep, model: string, usage: Usage | undefined): LlmUsage | null {
  if (!usage) return null;
  const line = usageOf(step, model, usage);
  const costUsd = line.costUsd === null ? null : Number(line.costUsd.toFixed(5));
  console.info('[leon]', 'llm.usage', JSON.stringify({ ...line, costUsd }));
  return line;
}
```

- [ ] **Étape 3 : écrire `record-usage.ts`**

`src/server/llm/record-usage.ts` :

```ts
import { resolveTenantContext } from '@/server/auth/tenant-context';
import { llmUsage } from '@/server/db/schema/usage';
import { withTenant } from '@/server/db/with-tenant';
import { logError } from '@/server/log';
import type { PipelineJob } from '@/server/pipeline/queue';
import { type Llm, type LlmRequest, type LlmUsage, PIPELINE_LLM_STEPS, type PipelineLlmStep } from './llm';

const isPipelineStep = (step: string): step is PipelineLlmStep =>
  (PIPELINE_LLM_STEPS as readonly string[]).includes(step);

// Enregistre l'usage d'un appel pour l'organisation et le run du job. Un échec est journalisé sans
// lever : l'appel est déjà payé, l'étape ne doit pas échouer pour une ligne de comptabilité.
export async function recordLlmUsage(job: PipelineJob, usage: LlmUsage): Promise<void> {
  const { step } = usage;
  if (!isPipelineStep(step)) return;
  try {
    const ctx = await resolveTenantContext(job.userId, job.tenantId);
    if (!ctx) return;
    await withTenant(ctx, (tx) =>
      tx.insert(llmUsage).values({
        tenantId: ctx.tenantId,
        ingestRunId: job.ingestRunId,
        step,
        model: usage.model,
        inputTokens: usage.inputTokens,
        outputTokens: usage.outputTokens,
        cacheReadTokens: usage.cacheReadTokens,
        cacheWriteTokens: usage.cacheWriteTokens,
        costUsd: usage.costUsd === null ? null : usage.costUsd.toFixed(8),
      }),
    );
  } catch (error) {
    logError('llm.usage_record_failed', error, { runId: job.ingestRunId });
  }
}

// Le client d'un job : chaque appel enregistre son usage, sans que le code des étapes le sache.
export function recordingUsage(llm: Llm, job: PipelineJob): Llm {
  return {
    generateObject<T>(request: LlmRequest<T>): Promise<T> {
      return llm.generateObject({ ...request, onUsage: (usage) => recordLlmUsage(job, usage) });
    },
  };
}
```

```ts
// Le worker du pipeline ne recharge pas le code à chaud : le commit au démarrage du serveur dit quel
// code traite les runs. Le jeu d'essai refuse de lancer un passage payant s'il diffère de HEAD.
const globalForVersion = globalThis as typeof globalThis & { leonStartCommit?: string | null };

export function readGitCommit(): string | null {
  try {
    return execFileSync('git', ['rev-parse', 'HEAD'], { encoding: 'utf8' }).trim();
  } catch {
    return null;
  }
}

export function recordStartCommit(): void {
  globalForVersion.leonStartCommit ??=
    process.env.NODE_ENV === 'production' ? null : readGitCommit();
}

export function startCommit(): string | null {
  return globalForVersion.leonStartCommit ?? null;
```

```ts
import { join } from 'node:path';
import { fileURLToPath } from 'node:url';
import type { z } from 'zod';
import { type IngestPayload, IngestPayloadSchema } from '@/capture/turn';
import { type Attendu, AttenduSchema, type Entreprise, EntrepriseSchema, MiniSchema } from './schema';

export const ESSAI_DIR = fileURLToPath(new URL('../../essai', import.meta.url));

export type ConversationEssai = {
  cle: string; // « hugo/2026-09-22-migration »
  personnage: string;
  payload: IngestPayload; // charge utile exacte de POST /api/v1/ingest
  attendu: Attendu;
};

export type JeuEssai = { entreprise: Entreprise; conversations: ConversationEssai[]; mini: string[] };

// Le message nomme le fichier, jamais son contenu.
async function readJson<T>(dir: string, path: string, schema: z.ZodType<T>): Promise<T> {
  let raw: string;
  try {
    raw = await readFile(join(dir, path), 'utf8');
  } catch {
    throw new Error(`Fichier introuvable : ${path}`);
  }
  const parsed = schema.safeParse(JSON.parse(raw));
  if (!parsed.success) throw new Error(`Fichier invalide : ${path} (${parsed.error.issues[0]?.path.join('.')})`);
  return parsed.data;
}

export async function loadJeu(dir = ESSAI_DIR): Promise<JeuEssai> {
  const entreprise = await readJson(dir, 'entreprise.json', EntrepriseSchema);
  const mini = await readJson(dir, 'mini.json', MiniSchema);
  const conversations: ConversationEssai[] = [];
  for (const personnage of (await readdir(join(dir, 'conversations'))).sort()) {
    const files = (await readdir(join(dir, 'conversations', personnage))).filter((f) => f.endsWith('.json'));
    for (const file of files.sort()) {
      const cle = `${personnage}/${file.slice(0, -'.json'.length)}`;
      conversations.push({
        cle,
        personnage,
        payload: await readJson(dir, `conversations/${cle}.json`, IngestPayloadSchema),
        attendu: await readJson(dir, `attendu/${cle}.json`, AttenduSchema),
      });
    }
  }
  return { entreprise, conversations, mini };
}

export const toursDe = (conversations: ConversationEssai[]) =>
  conversations.reduce((sum, c) => sum + c.payload.turns.length, 0);
```

```ts
import type { ConversationEssai, JeuEssai } from './dataset';
import { CLE_CONVERSATION } from './schema';

// Contrôles du jeu, sans appel : chaque problème est une ligne lisible, liste vide si tout va bien.
export function verifierJeu(jeu: JeuEssai): string[] {
  const problemes: string[] = [];
  const personnages = new Set(jeu.entreprise.personnages.map((p) => p.cle));
  const faitsConnus = new Set<string>();
  const vus = new Set<string>();
  const sources = new Set<string>();

  for (const c of jeu.conversations) {
    const ici = (message: string) => problemes.push(`${c.cle} : ${message}`);
    if (!CLE_CONVERSATION.test(c.cle)) ici('nom de fichier hors convention (ASCII, AAAA-MM-JJ-sujet)');
    if (!personnages.has(c.personnage)) ici(`personnage inconnu « ${c.personnage} »`);
    if (sources.has(c.payload.source.external_id)) ici('external_id de source en double');
    sources.add(c.payload.source.external_id);
    verifierTours(c, ici);
    for (const id of [...c.attendu.faits.map((f) => f.id), ...c.attendu.pasRetenir.map((n) => n.id)]) {
      if (vus.has(id)) problemes.push(`${id} défini deux fois`);
      vus.add(id);
      if (id.startsWith('F')) faitsConnus.add(id);
    }
  }

  for (const c of jeu.conversations) {
    for (const r of c.attendu.remplace)
      for (const id of [r.ancien, r.nouveau])
        if (!faitsConnus.has(id)) problemes.push(`${c.cle} : remplacement vers un fait inconnu ${id}`);
    for (const k of c.attendu.conflits)
      for (const id of [k.a, k.b])
        if (!faitsConnus.has(id)) problemes.push(`${c.cle} : conflit avec un fait inconnu ${id}`);
  }

  const cles = new Set(jeu.conversations.map((c) => c.cle));
  for (const cle of jeu.mini)
    if (!cles.has(cle)) problemes.push(`mini.json : conversation inconnue ${cle}`);

  const textes = jeu.conversations.flatMap((c) => c.payload.turns.map((t) => t.text));
  for (const secret of jeu.entreprise.secrets)
    if (!textes.some((t) => t.includes(secret)))
      problemes.push(`secret absent de toutes les conversations : ${secret.slice(0, 4)}…`);

  return problemes;
}

function verifierTours(c: ConversationEssai, ici: (message: string) => void): void {
  const tours = new Map(c.payload.turns.map((t) => [t.external_id, t]));
  if (tours.size !== c.payload.turns.length) ici('external_id de tour en double');
  const dates = c.payload.turns.map((t) => Date.parse(t.ts));
  if (dates.some((d, i) => i > 0 && d < (dates[i - 1] ?? d))) ici('tours hors ordre chronologique');
  for (const f of c.attendu.faits) {
    const tour = tours.get(f.tour);
    if (!tour) ici(`${f.id}, tour inconnu ${f.tour}`);
    else if (!tour.text.includes(f.extrait)) ici(`${f.id}, extrait absent du tour ${f.tour}`);
  }
  for (const n of c.attendu.pasRetenir)
    if (!tours.has(n.tour)) ici(`${n.id}, tour inconnu ${n.tour}`);
}

export function volumes(jeu: JeuEssai) {
  const parPersonnage = new Map<string, { personnage: string; conversations: number; tours: number; faits: number; pasRetenir: number }>();
  for (const c of jeu.conversations) {
    const v = parPersonnage.get(c.personnage) ?? { personnage: c.personnage, conversations: 0, tours: 0, faits: 0, pasRetenir: 0 };
    v.conversations += 1;
    v.tours += c.payload.turns.length;
    v.faits += c.attendu.faits.length;
    v.pasRetenir += c.attendu.pasRetenir.length;
    parPersonnage.set(c.personnage, v);
  }
  return [...parPersonnage.values()];
}
```

```ts
export type Compte = { cle: string; userId: string; token: string };

// Supprime l'organisation d'essai (sa mémoire, ses jetons et son usage suivent en cascade), puis ses
// comptes, et recrée le tout. Bornée au slug et au domaine d'essai, fixés par EntrepriseSchema.
export async function remettreAZero(
  db: AdminDb,
  entreprise: Entreprise,
): Promise<{ tenantId: string; comptes: Compte[] }> {
  if (entreprise.slug !== ESSAI_SLUG || entreprise.domaine !== ESSAI_DOMAINE)
    throw new Error('Remise à zéro refusée : ce n est pas l organisation d essai.');
  const password = await hashPassword(entreprise.motDePasse);
  return db.transaction(async (tx) => {
    // L'organisation d'abord : les faits référencent leurs auteurs sans cascade.
    await tx.delete(organizations).where(eq(organizations.slug, ESSAI_SLUG));
    await tx.delete(users).where(like(users.email, `%@${ESSAI_DOMAINE}`));
    const [org] = await tx
      .insert(organizations)
      .values({ name: entreprise.organisation, slug: ESSAI_SLUG, createdAt: new Date() })
      .returning({ id: organizations.id });
    if (!org) throw new Error('organisation d essai non créée');

    const comptes: Compte[] = [];
    for (const [index, personnage] of entreprise.personnages.entries()) {
      const [user] = await tx
        .insert(users)
        .values({ name: personnage.nom, email: `${personnage.cle}@${ESSAI_DOMAINE}`, emailVerified: true })
        .returning({ id: users.id });
      if (!user) throw new Error('compte d essai non créé');
      // Compte « credential » de Better Auth : connexion par e-mail et mot de passe dans l'interface.
      await tx.execute(sql`insert into account ("accountId", "providerId", "userId", password, "createdAt", "updatedAt")
                           values (${user.id}, 'credential', ${user.id}, ${password}, now(), now())`);
      await tx.insert(members).values({
        organizationId: org.id,
        userId: user.id,
        role: index === 0 ? 'owner' : 'member',
        createdAt: new Date(),
      });
      const token = generateApiToken();
      await tx.insert(apiTokens).values({
        tenantId: org.id,
        userId: user.id,
        name: 'Jeu d essai',
        tokenHash: hashApiToken(token),
        tokenHint: token.slice(-4),
      });
      comptes.push({ cle: personnage.cle, userId: user.id, token });
    }
    return { tenantId: org.id, comptes };
  });
}

}

    .from(ingestRuns)

}
```

```ts
import { type IngestPayload, type IngestResult, TURNS_PER_BATCH_MAX } from '@/capture/turn';
import type { ConversationEssai } from './dataset';

const debut = (c: ConversationEssai) => Date.parse(c.payload.turns[0]?.ts ?? '');

// Ordre chronologique global : la mémoire se construit dans l'ordre où l'entreprise a parlé.
export function ordreEnvoi(conversations: ConversationEssai[]): ConversationEssai[] {
  return [...conversations].sort((a, b) => debut(a) - debut(b) || a.cle.localeCompare(b.cle));
}

export function lots(payload: IngestPayload, max = TURNS_PER_BATCH_MAX): IngestPayload[] {
  const result: IngestPayload[] = [];
  for (let i = 0; i < payload.turns.length; i += max)
    result.push({ source: payload.source, turns: payload.turns.slice(i, i + max) });
  return result;
}

export type Bilan = { envois: number; tours: number; inseres: number; runs: number };

// Un envoi à la fois, dans l'ordre : comme la capture, qui envoie au fil des conversations.
export async function envoyer(
  conversations: ConversationEssai[],
  jetons: Map<string, string>,
  send: (token: string, payload: IngestPayload) => Promise<IngestResult>,
  out: (line: string) => void,
): Promise<Bilan> {
  const bilan: Bilan = { envois: 0, tours: 0, inseres: 0, runs: 0 };
  for (const [index, c] of ordreEnvoi(conversations).entries()) {
    const token = jetons.get(c.personnage);
    if (!token) throw new Error(`Aucun jeton pour le personnage ${c.personnage}.`);
    for (const lot of lots(c.payload)) {
      const result = await send(token, lot);
      bilan.envois += 1;
      bilan.tours += lot.turns.length;
      bilan.inseres += result.inserted;
      if (result.ingestRunId) bilan.runs += 1;
    }
    out(`[${index + 1}/${conversations.length}] ${c.cle} (${c.payload.turns.length} tours)`);
  }
  return bilan;
}
```

```ts
  appels: number;
  inputTokens: number;
  outputTokens: number;
  cacheReadTokens: number;
  cacheWriteTokens: number;
  costUsd: number;

  const rows = await db
    .select({
      step: llmUsage.step,

      outputTokens: sql<number>`sum(${llmUsage.outputTokens})::int`,
      cacheReadTokens: sql<number>`sum(${llmUsage.cacheReadTokens})::int`,
      cacheWriteTokens: sql<number>`sum(${llmUsage.cacheWriteTokens})::int`,
      costUsd: sql<string>`coalesce(sum(${llmUsage.costUsd}), 0)`,
    })
    .from(llmUsage)
    .where(eq(llmUsage.tenantId, tenantId))
    .groupBy(llmUsage.step);

}
```

```ts
import { z } from 'zod';
import type { Llm, LlmUsage } from '@/server/llm/llm';
import type { FaitProduit } from './collect';
import type { ConversationEssai } from './dataset';

export const JugeOutputSchema = z.object({
  paires: z.array(z.object({ produit: z.string(), correspond: z.string().nullable() })),
});
type JugeOutput = z.infer<typeof JugeOutputSchema>;

const SYSTEM = `Tu compares des faits extraits automatiquement d'une conversation (identifiants P…) à une liste de référence : les faits attendus (F…) et les faits qu'il ne fallait pas retenir (N…).
Pour chaque fait P, donne l'identifiant F ou N qui porte la même information sur le même sujet, même formulée autrement ou de façon partielle ; null si aucun ne correspond.
Plusieurs P peuvent viser le même F. N'invente aucun identifiant. Réponds pour chaque P, une seule fois.`;

// Un identifiant inventé ou un P oublié vaut « aucune correspondance » : jamais de paire hors des listes.
export function validerPaires(output: JugeOutput, produits: string[], attendus: string[]): Map<string, string | null> {
  const connus = new Set(attendus);
  const donnees = new Map<string, string | null>();
  for (const { produit, correspond } of output.paires)
    if (!donnees.has(produit)) donnees.set(produit, correspond !== null && connus.has(correspond) ? correspond : null);
  return new Map(produits.map((id) => [id, donnees.get(id) ?? null]));
}

// Paires « identifiant du fait produit → F…, N… ou null » pour une conversation.
export async function juger(
  llm: Llm,
  conversation: ConversationEssai,
  produits: FaitProduit[],
  onUsage?: (usage: LlmUsage) => Promise<void>,
): Promise<Map<string, string | null>> {
  if (produits.length === 0) return new Map();
  const locaux = produits.map((f, i) => ({ local: `P${i + 1}`, fait: f }));
  const prompt = JSON.stringify(
    {
      attendus: conversation.attendu.faits.map((f) => ({ id: f.id, enonce: f.enonce, fiche: f.fiche })),
      aNePasRetenir: conversation.attendu.pasRetenir.map((n) => ({ id: n.id, enonce: n.enonce })),
      produits: locaux.map(({ local, fait }) => ({ id: local, enonce: fait.statement, fiche: fait.ficheTitre })),
    },
    null,
    2,
  );
  const output = await llm.generateObject({ step: 'juge', system: SYSTEM, prompt, schema: JugeOutputSchema, maxTokens: 4_000, onUsage });
  const attendus = [...conversation.attendu.faits.map((f) => f.id), ...conversation.attendu.pasRetenir.map((n) => n.id)];
  const paires = validerPaires(output, locaux.map((l) => l.local), attendus);
  return new Map(locaux.map(({ local, fait }) => [fait.id, paires.get(local) ?? null]));
}
```

```ts
import { normalizeStatement } from '@/server/pipeline/placement/plan';
import type { FaitProduit } from './collect';
import type { ConversationEssai } from './dataset';
import type { FaitAttendu } from './schema';

export type Ratio = { ok: number; sur: number };

export type NotesConversation = {
  cle: string;
  attendus: number;
  retrouves: number;
  produits: number;
  manques: string[]; // F… jamais retrouvés
  enTrop: string[]; // faits produits sans correspondance
  fauxFaits: string[]; // faits produits qui correspondent à un N…
};

export type Notes = {
  attendus: number;
  retrouves: number;
  rappel: number | null;
  produits: number;
  apparies: number;
  precision: number | null;
  fauxFaits: number;
  doublonsFaits: number;
  tourCorrect: Ratio;
  ficheCorrecte: Ratio;
  typeCorrect: Ratio;
  doublonsFiches: number;
  remplacements: { trouves: number; attendus: number };
  conflits: { trouves: number; attendus: number };
  sensibles: { marques: number; attendus: number; retrouves: number };
  sensiblesPartages: number;
  secretsFuites: number;
  parConversation: NotesConversation[];
};

const ratio = (a: number, b: number) => (b === 0 ? null : a / b);

function ficheAttendue(fait: FaitAttendu, titre: string): boolean {
  const titres = [fait.fiche, ...fait.alias].map(normalizeStatement);
  return titres.includes(normalizeStatement(titre));
}

// Notes d'un passage, sans LLM : les paires du juge (fait produit → F…, N… ou null) font foi.
export function calculerNotes(input: {
  conversations: ConversationEssai[];
  produits: FaitProduit[];
  paires: Map<string, string | null>;
  secrets: string[];
}): Notes {
  const { conversations, produits, paires } = input;
  const attendus = new Map(conversations.flatMap((c) => c.attendu.faits.map((f) => [f.id, f] as const)));
  const parAttendu = new Map<string, FaitProduit[]>();
  for (const produit of produits) {
    const cible = paires.get(produit.id);
    if (cible?.startsWith('F')) parAttendu.set(cible, [...(parAttendu.get(cible) ?? []), produit]);
  }
  const apparies = [...parAttendu.entries()].flatMap(([id, liste]) => liste.map((p) => ({ attendu: attendus.get(id), produit: p })));
  const compte = (test: (a: FaitAttendu, p: FaitProduit) => boolean): Ratio => ({
    ok: apparies.filter(({ attendu, produit }) => attendu && test(attendu, produit)).length,
    sur: apparies.length,
  });

  // Une fiche attendue rangée dans plusieurs fiches produites : chaque fiche en plus est un doublon.
  const fichesParAttendue = new Map<string, Set<string>>();
  for (const { attendu, produit } of apparies) {
    if (!attendu) continue;
    const cle = normalizeStatement(attendu.fiche);
    fichesParAttendue.set(cle, (fichesParAttendue.get(cle) ?? new Set()).add(produit.ficheId));
  }

  const statut = (id: string) => (parAttendu.get(id) ?? []).map((p) => p.status);
  const remplaces = conversations.flatMap((c) => c.attendu.remplace);
  const conflits = conversations.flatMap((c) => c.attendu.conflits);
  const sensiblesAttendus = [...attendus.values()].filter((f) => f.sensible);
  const sensiblesRetrouves = sensiblesAttendus.filter((f) => parAttendu.has(f.id));
  const nombreApparies = produits.filter((p) => paires.get(p.id)?.startsWith('F')).length;
  const retrouves = [...attendus.keys()].filter((id) => parAttendu.has(id)).length;

  return {
    attendus: attendus.size,
    retrouves,
    rappel: ratio(retrouves, attendus.size),
    produits: produits.length,
    apparies: nombreApparies,
    precision: ratio(nombreApparies, produits.length),
    fauxFaits: produits.filter((p) => paires.get(p.id)?.startsWith('N')).length,
    doublonsFaits: [...parAttendu.values()].reduce((sum, liste) => sum + liste.length - 1, 0),
    tourCorrect: compte((a, p) => a.tour === p.turnExternalId),
    ficheCorrecte: compte((a, p) => ficheAttendue(a, p.ficheTitre)),
    typeCorrect: compte((a, p) => a.typeFiche === p.typeFiche),
    doublonsFiches: [...fichesParAttendue.values()].reduce((sum, fiches) => sum + fiches.size - 1, 0),
    remplacements: {
      trouves: remplaces.filter((r) => statut(r.ancien).includes('superseded') && statut(r.nouveau).includes('active')).length,
      attendus: remplaces.length,
    },
    conflits: {
      trouves: conflits.filter((k) => [...statut(k.a), ...statut(k.b)].includes('conflict')).length,
      attendus: conflits.length,
    },
    sensibles: {
      marques: sensiblesRetrouves.filter((f) => (parAttendu.get(f.id) ?? []).every((p) => p.sensitivity === 'sensitive')).length,
      attendus: sensiblesAttendus.length,
      retrouves: sensiblesRetrouves.length,
    },
    sensiblesPartages: produits.filter((p) => p.sensitivity === 'sensitive' && p.partage).length,
    secretsFuites: produits.filter((p) => input.secrets.some((s) => [p.statement, p.quote, p.value ?? ''].some((t) => t.includes(s)))).length,
    parConversation: conversations.map((c) => notesConversation(c, produits, paires, parAttendu)),
  };
}

function notesConversation(
  c: ConversationEssai,
  produits: FaitProduit[],
  paires: Map<string, string | null>,
  parAttendu: Map<string, FaitProduit[]>,
): NotesConversation {
  const siens = produits.filter((p) => p.sourceExternalId === c.payload.source.external_id);
  const manques = c.attendu.faits.filter((f) => !parAttendu.has(f.id)).map((f) => f.id);
  return {
    cle: c.cle,
    attendus: c.attendu.faits.length,
    retrouves: c.attendu.faits.length - manques.length,
    produits: siens.length,
    manques,
    enTrop: siens.filter((p) => !paires.get(p.id)).map((p) => p.id),
    fauxFaits: siens.filter((p) => paires.get(p.id)?.startsWith('N')).map((p) => p.id),
  };
}
```

```ts
import { appendFile, mkdir, readdir, readFile, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { fileURLToPath } from 'node:url';
import type { MODEL_BY_STEP } from '@/server/llm/anthropic-llm';
import type { etatRuns, UsageParEtape } from './collect';
import type { Notes } from './score';

export const HISTORIQUE_DIR = fileURLToPath(new URL('../../docs/etudes/essais-pipeline', import.meta.url));

export type Passage = {
  id: string; // « 2026-10-16-1000-reference » : nom du fichier JSON
  date: string;
  label: string;
  commit: string;
  jeu: 'complet' | 'mini';
  modeles: Record<keyof typeof MODEL_BY_STEP, string>;
  tours: number;
  dureeMinutes: number | null;
  runs: Awaited<ReturnType<typeof etatRuns>>;
  usage: UsageParEtape; // mesuré (llm_usage)
  coutJuge: number; // mesuré, hors coût du pipeline
  notes: Notes;
};

const EN_TETE = `# Historique des essais du pipeline

Un passage par ligne, ajouté par \`pnpm essai noter\`. Coûts **mesurés** (table \`llm_usage\`, tarifs de \`docs/etudes/couts-2026-10/annexes/01-prix-llm.md\`), hors coût du juge. Détail de chaque passage dans \`passages/\`.

| Date | Libellé | Commit | Jeu | Tours | Coût pipeline | Coût / tour | Rappel | Précision | Faux faits | Tour juste | Fiche juste | Fiches en double | Remplacements | Conflits | Sensibles | Secrets | Runs échoués | Durée |
|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|---|
`;

const euros = (n: number, d = 2) => `${n.toFixed(d).replace('.', ',')} $`;
const pct = (n: number | null) => (n === null ? 'n/d' : `${Math.round(n * 100)} %`);
const sur = (a: number, b: number) => `${a}/${b}`;

export const coutPipeline = (usage: UsageParEtape) => Object.values(usage).reduce((sum, u) => sum + u.costUsd, 0);

export function ligneHistorique(p: Passage): string {
  const n = p.notes;
  const cout = coutPipeline(p.usage);
  return `| ${[
    p.date.slice(0, 10),
    p.label,
    p.commit.slice(0, 7),
    p.jeu,
    p.tours,
    euros(cout),
    euros(cout / Math.max(p.tours, 1), 4),
    pct(n.rappel),
    pct(n.precision),
    n.fauxFaits,
    pct(n.tourCorrect.sur ? n.tourCorrect.ok / n.tourCorrect.sur : null),
    pct(n.ficheCorrecte.sur ? n.ficheCorrecte.ok / n.ficheCorrecte.sur : null),
    n.doublonsFiches,
    sur(n.remplacements.trouves, n.remplacements.attendus),
    sur(n.conflits.trouves, n.conflits.attendus),
    `${sur(n.sensibles.marques, n.sensibles.retrouves)}, ${n.sensiblesPartages} partagé(s)`,
    n.secretsFuites,
    p.runs.failed,
    p.dureeMinutes === null ? 'n/d' : `${p.dureeMinutes} min`,
  ].join(' | ')} |\n`;
}

export async function enregistrerPassage(p: Passage, dir = HISTORIQUE_DIR): Promise<string> {
  await mkdir(join(dir, 'passages'), { recursive: true });
  const md = join(dir, 'historique.md');
  const existe = await readFile(md, 'utf8').then(() => true, () => false);
  await appendFile(md, `${existe ? '' : EN_TETE}${ligneHistorique(p)}`);
  const json = join(dir, 'passages', `${p.id}.json`);
  await writeFile(json, `${JSON.stringify(p, null, 2)}\n`);
  return json;
}

// Coût mesuré par tour du dernier passage complet, pour annoncer le coût du suivant.
export async function coutParTourPrecedent(dir = HISTORIQUE_DIR): Promise<number | null> {
  const fichiers = await readdir(join(dir, 'passages')).catch(() => [] as string[]);
  for (const fichier of fichiers.filter((f) => f.endsWith('.json')).sort().reverse()) {
    const p = JSON.parse(await readFile(join(dir, 'passages', fichier), 'utf8')) as Passage;
    if (p.jeu === 'complet' && p.tours > 0) return coutPipeline(p.usage) / p.tours;
  }
  return null;
}
```

```ts
  label.normalize('NFD').replace(/\p{M}/gu, '').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '') || 'essai';

async function runVerifier(): Promise<number> {
  const jeu = await loadJeu();
  console.table(volumes(jeu));

async function runCharger(args: string[]): Promise<number> {
  const { values } = parseArgs({ args, options: { mini: { type: 'boolean' }, label: { type: 'string' }, oui: { type: 'boolean' } } });
  const jeu = await loadJeu();
  const problemes = verifierJeu(jeu);
  if (problemes.length > 0) throw new Error(`Jeu invalide (${problemes.length} problème(s)) : lance « pnpm essai verifier ».`);
  const conversations = values.mini ? jeu.conversations.filter((c) => jeu.mini.includes(c.cle)) : jeu.conversations;

  const head = gitHead();
  const refus = await verifierServeur({ fetch, url: leonUrl(), head, srcModifie: srcModifie() });
  if (refus) throw new Error(refus);

  const admin = openAdminDb();
  try {
    if ((await runsEnCours(admin.db)) > 0) throw new Error('Des runs du passage précédent tournent encore : attends-les (pnpm essai noter).');
    const tours = toursDe(conversations);
    const parTour = (await coutParTourPrecedent()) ?? COUT_PAR_TOUR_14_10;
    console.log(`${conversations.length} conversations, ${tours} tours. Coût estimé : ${(tours * parTour).toFixed(2)} $ (${parTour.toFixed(4)} $ par tour).`);
    if (!values.oui) {
      const rl = createInterface({ input: process.stdin, output: process.stdout });
      const reponse = await rl.question('Lancer ? (o/N) ');
      rl.close();
      if (reponse.trim().toLowerCase() !== 'o') return 1;
    }

    const { comptes } = await remettreAZero(admin.db, jeu.entreprise);
    const label = values.label ?? 'sans libellé';
    const debut = new Date();
    const passage: DernierPassage = {

      id: `${debut.toISOString().slice(0, 16).replace('T', '-').replace(':', '')}-${slugLabel(label)}`,

      commit: head,
      jeu: values.mini ? 'mini' : 'complet',
      conversations: conversations.map((c) => c.cle),
      tours,
      debut: debut.toISOString(),
    };
    await writeFile(DERNIER, `${JSON.stringify(passage, null, 2)}\n`);

    const jetons = new Map(comptes.map((c) => [c.cle, c.token]));
    const bilan = await envoyer(conversations, jetons, (token, payload) => createIngestSender(fetch, { url: leonUrl(), token })(payload), console.log);
    console.log(`Envoyé : ${bilan.tours} tours en ${bilan.envois} envois, ${bilan.runs} runs. Mot de passe des personnages : essai/README.md.`);

}

async function attendreRuns(db: ReturnType<typeof openAdminDb>['db'], tenantId: string) {
  for (;;) {
    const etat = await etatRuns(db, tenantId);
    process.stdout.write(`\rRuns : ${etat.done} terminés, ${etat.failed} échoués, ${etat.enCours} en cours sur ${etat.total}   `);
    if (etat.enCours === 0) {
      process.stdout.write('\n');
      return etat;
    }
    await new Promise((resolve) => setTimeout(resolve, 10_000));
  }
}

async function runNoter(): Promise<number> {
  const passage = DernierPassageSchema.parse(JSON.parse(await readFile(DERNIER, 'utf8')));
  const apiKey = process.env.ANTHROPIC_API_KEY;
  if (!apiKey) throw new Error('ANTHROPIC_API_KEY manquante (le juge en a besoin).');
  const jeu = await loadJeu();
  const conversations = jeu.conversations.filter((c) => passage.conversations.includes(c.cle));
  const admin = openAdminDb();
  try {
    const tenantId = await tenantEssai(admin.db);
    if (!tenantId) throw new Error('Organisation d essai introuvable : lance « pnpm essai charger ».');
    const runs = await attendreRuns(admin.db, tenantId);
    const produits = await lireFaits(admin.db, tenantId);
    const usage = await lireUsage(admin.db, tenantId);

    const llm = createAnthropicLlm({ apiKey });
    let coutJuge = 0;
    const compter = async (u: LlmUsage) => {
      coutJuge += u.costUsd ?? 0;
    };
    const paires = new Map<string, string | null>();
    for (const c of conversations) {
      const siens = produits.filter((p) => p.sourceExternalId === c.payload.source.external_id);
      for (const [id, cible] of await juger(llm, c, siens, compter)) paires.set(id, cible);
    }

    const notes = calculerNotes({ conversations, produits, paires, secrets: jeu.entreprise.secrets });
    const fin = runs.fin ? new Date(runs.fin) : null;
    const json = await enregistrerPassage({
      id: passage.id,
      date: passage.debut,
      label: passage.label,
      commit: passage.commit,
      jeu: passage.jeu,
      modeles: { ...MODEL_BY_STEP },
      tours: passage.tours,
      dureeMinutes: fin ? Math.round((fin.getTime() - Date.parse(passage.debut)) / 60_000) : null,
      runs,
      usage,
      coutJuge,
      notes,
    });
    console.log(`Rappel ${Math.round((notes.rappel ?? 0) * 100)} %, précision ${Math.round((notes.precision ?? 0) * 100)} %, ${notes.fauxFaits} faux faits, ${runs.failed} run(s) échoué(s).`);
    console.log(`Coût du pipeline : ${Object.values(usage).reduce((s, u) => s + u.costUsd, 0).toFixed(3)} $ ; juge : ${coutJuge.toFixed(3)} $.`);
    console.log(`Historique mis à jour ; détail dans ${json}.`);
    return 0;
  } finally {
    await admin.close();
  }
}

async function main(argv: string[]): Promise<number> {
  const [command, ...args] = argv;
  try {
    if (command === 'verifier') return await runVerifier();
    if (command === 'charger') return await runCharger(args);
    if (command === 'noter') return await runNoter();
  } catch (error) {
    console.error(error instanceof Error ? error.message : 'Erreur inconnue.');
    return 1;
  }
  const asked = command === undefined || command === 'help' || command === '--help';
  (asked ? console.log : console.error)(HELP);
  return asked ? 0 : 1;
}

process.exitCode = await main(process.argv.slice(2));
```
